static function ($val) {
    $str = '';
    if (Perl::num($val) >= 86400) {
        $d = Perl::int(Perl::num($val) / 86400);
        $str .= Perl::str($d) . ' days ';
        $val = Perl::num($val) - $d * 24 * 3600;
    }
    $h = Perl::int(Perl::num($val) / 3600);
    $val = Perl::num($val) - $h * 3600;
    $m = Perl::int($val / 60);
    $val -= $m * 60;
    $ss = Perl::sprintf('%05.2f', $val);
    if (Perl::num($ss) >= 60) {
        $ss = '00.00';
        if (++$m >= 60) {
            $m -= 60;
            ++$h;
        }
    }
    return Perl::sprintf('%s%.2d:%.2d:%s', $str, $h, $m, $ss);
}
