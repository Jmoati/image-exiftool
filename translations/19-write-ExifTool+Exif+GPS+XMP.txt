# Extra Geolocate
static function ($val, ExifTool $self, $wantGroup) {
    $args = Perl::split('/\s*,\s*/', $val);
    $lat = 1;
    foreach ($args as &$_) {
        if (!preg_match('/^[-+]?\d/', $_)) {
            continue;
        }
        $reals = preg_match_all('/\.\d+/', $_);
        if ($reals > 1) {
            continue; // (allow floating "lat lon" format)
        }
        ExifTool::RequireModule('Image::ExifTool::GPS');
        $_ = GPS::ToDegrees($_, 1, $lat ? 'lat' : 'lon');
        $lat ^= 1;
    }
    unset($_);
    return implode(',', array_map(Perl::str(...), $args));
}

==== afdc3e5467 ValueConvInv
# Extra Geolocate
static function ($val, ExifTool $self, $wantGroup) {
    ExifTool::RequireModule('Image::ExifTool::Geolocation');
    // write this tag later if geotagging
    if (preg_match('/\bgeotag\b/i', Perl::str($val))) {
        return $val;
    }
    $val = Perl::str($val).',both';
    $opts = $self->OPTIONS;
    [$cities, $dist] = Geolocation::Geolocate($self->Encode($val, 'UTF8'), $opts);
    if (!$cities) {
        return '';
    }
    if (count($cities) > 1 and $self->Warn('Multiple matching cities found', 2)) {
        trigger_error(Perl::strw($self->VALUE['Warning'] ?? null, 'concatenation (.) or string')."\n", E_USER_WARNING);
        return '';
    }
    $geo = Geolocation::GetEntry($cities[0], $opts['Lang'] ?? null);
    $tags = $self->GetGeolocateTags($wantGroup, $dist ? 0 : 1);
    $geoNum = ['City' => 0, 'Province' => 1, 'State' => 1, 'Code' => 3, 'Country' => 4,
               'Coordinates' => 89, 'Latitude' => 8, 'Longitude' => 9];
    $value = null;
    foreach ($tags as $tag) {
        if (preg_match('/GPS(Coordinates|Latitude|Longitude)?/', $tag, $m)) {
            // (all the GPS tags of GetGeolocateTags end with one of these words)
            $n = $geoNum[$m[1] ?? ''] ?? null;
            $value = $n == 89 ? Perl::str($geo[8] ?? null).','.Perl::str($geo[9] ?? null) : ($geo[(int) $n] ?? null);
        } elseif (preg_match('/(Code)/', $tag, $m) or preg_match('/(City|Province|State|Country)/', $tag, $m)) {
            $value = $geo[$geoNum[$m[1]]] ?? null;
            if ($value === null) {
                continue;
            }
            $value = $self->Decode($value, 'UTF8');
            if ($tag === 'iptc:Country-PrimaryLocationCode') {
                $value .= ' '; // (IPTC requires 3-char code)
            }
        } elseif (preg_match('/LocationName/', $tag)) {
            $value = $geo[0] ?? null;
            if (!$value) {
                continue;
            }
            if ($geo[1] ?? null) {
                $value .= ', '.$geo[1];
            }
            if ($geo[4] ?? null) {
                $value .= ', '.$geo[4];
            }
            $value = $self->Decode($value, 'UTF8');
        } else {
            continue; // (shouldn't happen)
        }
        $self->SetNewValue($tag, $value, Type: 'PrintConv');
    }
    return '';
}

static function ($str) {
    [$bit, $val, $str] = [8, 0, Perl::str($str)];
    if (strlen($str) == 10) {
        $str = substr($str, 1);
    }
    if (strlen($str) != 9) {
        return null;
    }
    while ($bit >= 0) {
        foreach (['r', 'w', 'x'] as $_) {
            if (substr($str, 8 - $bit, 1) === $_) {
                $val |= 1 << $bit;
            }
            --$bit;
        }
    }
    return Perl::sprintf('%.3o', $val);
}

==== 24785b319e PrintConvInv
# Exif::Main 0xc763 TimeCodes
static function ($val, ExifTool $self, $wantGroup) {
    $a = Perl::split(' ', $val);
    $v = [];
    foreach ($a as $_) {
        $td = array_reverse(Perl::split('/T/', $_));
        $tz = 0x39; // default to unknown timezone
        $td0 = Perl::strw($td[0] ?? null, 'substitution (s///)', '$td[0]');
        if (preg_match('/([-+])(\d+):(\d+)$/', $td0, $m)) {
            $td[0] = preg_replace('/([-+])(\d+):(\d+)$/', '', $td0, 1);
            if ($m[3] == 0) {
                $tz = Perl::hexw(($m[1] === '-') ? $m[2] : 0x26 - (int) $m[2]);
            } elseif ($m[3] == 30) {
                if ($m[1] === '-') {
                    $tz = Perl::num($m[2]) + 0x0a;
                    if ($tz > 0x0f) {
                        $tz += 0x0a;
                    }
                } else {
                    $tz = 0x3f - Perl::num($m[2]);
                    if ($tz < 0x3a) {
                        $tz -= 0x0a;
                    }
                }
            } elseif ($m[3] == 45) {
                if ($m[1] === '+' and $m[2] == 12) {
                    $tz = 0x32;
                }
            }
        }
        $t = Perl::split('/[:.]/', Perl::strw($td[0] ?? null, 'split', '$td[0]'));
        while (count($t) < 4) {
            $t[] = '00';
        }
        if ($td[1] ?? null) {
            // date was specified: fill in date & timezone
            $d = Perl::split('/[-]/', $td[1]);
            if (count($d) < 3) {
                continue;
            }
            $bg = Perl::sprintf('.%.2d.%.2d.%.2d.%.2x', Perl::numw($d[2], 'sprintf'), Perl::numw($d[1], 'sprintf'),
                Perl::mod(Perl::numw($d[0], 'modulus (%)', '$d[0]'), 100), $tz);
            $t[0] = Perl::sprintf('%.2x', Perl::hexw($t[0], '$t[0]') + 0xc0); // set BGF1+BGF2
        } else { // time only
            $bg = '.00.00.00.00';
        }
        $v[] = implode('.', array_reverse(array_slice($t, 0, 4))).$bg;
    }
    return implode(' ', $v);
}

        $s = $m[1] === '-' ? 1 : -1;
        $t = $m[2];

    }
    $a = preg_match('/^[^\d]*\d{4}[^\d]*\d{1,2}[^\d]*\d{1,2}[^\d]*(\d{1,2})[^\d]*(\d{2})[^\d]*(\d{2}(?:\.\d+)?)[^\d]*$/', $v, $m) ?
