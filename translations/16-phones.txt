==== 0c90ebb73d RawConv
# PLIST::Main XMLFileType
static function ($val, ExifTool $self, $tag, ?TagInfo $tagInfo, &$priority, array &$grps, array $vals) {
    if (Perl::str($val) === 'ModdXML' && Perl::str($self->FILE_TYPE) === 'XMP') {
        $self->OverrideFileType('MODD');
    }
    return $val;
}
