==== 58e93a5ea7 ValueConv
# MPEG::Composite Duration Duration
static function ($val, ExifTool $self, array &$vals, array $prt, array $raw, ?TagInfo $tagInfo, $tag) {
    if (($vals[4] ?? null) && ($vals[5] ?? null) !== null && ($vals[6] ?? null) !== null) {
        // calculate from number of VBR audio frames
        $mfs = Perl::num($prt[5] ?? null) / (Perl::num($vals[6]) == 3 ? 144 : 72);
        // calculate using VBR length
        return 8 * Perl::num($vals[4]) / $mfs;
    }
    // calculate duration as file size divided by total bitrate
    // (note: this is only approximate!)
    if (!(($vals[2] ?? null) || ($vals[3] ?? null))) return null;
    if (($vals[2] ?? null) && !preg_match('/^\d+$/', Perl::str($vals[2]))) return null;
    if (($vals[3] ?? null) && !preg_match('/^\d+$/', Perl::str($vals[3]))) return null;
    return (8 * (Perl::num($vals[0] ?? null) - Perl::num(($vals[1] ?? null) ?: 0))) / (Perl::num(($vals[2] ?? null) ?: 0) + Perl::num(($vals[3] ?? null) ?: 0));
}

==== 641c610afa PrintConv
# MacOS::XAttr com.apple.FinderInfo XAttrFinderInfo
static function ($val, ExifTool $self, array &$vals, array $prt, array $raw, ?TagInfo $tagInfo, $tag) {
    $val = Perl::str($val);
    if (!preg_match("/^('.*?') ('.*?') /s", $val, $m)) return $val;
    $val = substr($val, \strlen($m[0]));
    [$type, $creator] = [$m[1], $m[2]];
    [$flags, $y, $x, $exFlags, $putAway] = array_pad(Perl::split(' ', $val), 5, null);
    $label = (Perl::int($flags) >> 1) & 0x07;
    $flags = ExifTool::DecodeBits(((Perl::int($exFlags) << 16) | Perl::int($flags)) & 0xfff1, [
        0 => 'OnDesk',
        6 => 'Shared',
        7 => 'HasNoInits',
        8 => 'Inited',
        10 => 'CustomIcon',
        11 => 'Stationery',
        12 => 'NameLocked',
        13 => 'HasBundle',
        14 => 'Invisible',
        15 => 'Alias',
        // extended flags
        22 => 'HasRoutingInfo',
        23 => 'ObjectBusy',
        24 => 'CustomBadge',
        31 => 'ExtendedFlagsValid',
    ]);
    $str = "Type=$type Creator=$creator Flags=" . Perl::str($flags) . " Label=$label Pos=(" . Perl::str($x) . ',' . Perl::str($y) . ')';
    if ($putAway) $str .= ' Putaway=' . Perl::str($putAway);
    return $str;
}
