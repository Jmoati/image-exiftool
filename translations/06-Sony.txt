# Composite RawConv: @val holds the raw values of the Require'd tags (HiddenDataOffset, HiddenDataLength).
# Sony::ReadHiddenData($et, &$hdOff, $hdLen) must take $hdOff by reference (Perl sets $_[1] to the fixed offset).
# It returns \$buff in Perl, kept as the binary tag value: accept the data string or a ScalarRef.
static function ($val, ExifTool $self, $tag, ?TagInfo $tagInfo, &$priority, array &$grps, array $vals) {

        return null;
    }
    $dataPt = Sony::ReadHiddenData($self, $hdOff, $vals[1] ?? null);
    if ($dataPt !== null && $hDump) {
        $msg = '(Sony HiddenData)';
        if (Perl::num($hdOff) != Perl::num($vals[0] ?? null)) {
            $msg .= ' <span class=V>(fixed)</span>';

        $self->HDump($hdOff, $vals[1] ?? null, $msg, null, 0x08);

}
