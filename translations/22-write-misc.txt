==== 967a809307 PrintConvInv
# CanonCustom::Functions2 0x010c ShutterSpeedRange
# $val=~m{([\d./]+)} ? eval $1 : 0  (eval of Perl code made of digits, dots and slashes: a
# v-string, or numbers divided left to right, where a leading 0 makes an octal integer, which a
# following "." concatenates; undef when the eval fails)
static function ($val, ExifTool $self, $wantGroup) {
    if (!preg_match('{([\d./]+)}', Perl::str($val), $m)) {
        return 0;
    }
    $e = $m[1];
    if (preg_match('/^\d+\.\d+(\.\d+)+$/', $e)) {
        return implode('', array_map(static fn ($n) => Perl::chr((int) $n), explode('.', $e)));
    }
    // terms joined by ".", each one numbers joined by "/"
    $terms = [];
    $pos = 0;
    $len = strlen($e);
    while ($pos < $len) {
        $nums = [];
        for (;;) {
            if (!preg_match('{\G(?:(0\d+)|(\d+\.?\d*|\.\d+))}', $e, $t, 0, $pos)) {
                return null;    // (syntax error)
            }
            $pos += strlen($t[0]);
            if (($t[1] ?? '') !== '') {
                if (preg_match('/[89]/', $t[1])) {
                    return null;    // (Illegal octal digit)
                }
                $nums[] = octdec($t[1]);
            } else {
                $nums[] = Perl::num($t[2]);
            }
            if ($pos < $len && $e[$pos] === '/') {
                ++$pos;
                continue;
            }
            break;
        }
        $r = array_shift($nums);
        foreach ($nums as $n) {
            if ($n == 0) {
                return null;    // (Illegal division by zero)
            }
            $r = $r / $n;
        }
        $terms[] = $r;
        if ($pos < $len) {
            if ($e[$pos] !== '.') {
                return null;
            }
            ++$pos;
        }
    }
    return count($terms) == 1 ? $terms[0] : implode('', array_map(Perl::str(...), $terms));
}
